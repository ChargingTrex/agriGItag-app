# verifier script source: accept.py
"""W01 verifier: independent acceptance checks over a built output dir. Usage: accept.py <out_dir> <fixture.json>"""
import json, re, sys
from pathlib import Path
import html5lib
from html5lib.html5parser import ParseError

out, fx = Path(sys.argv[1]), json.loads(Path(sys.argv[2]).read_text())
html = (out / "index.html").read_text(encoding="utf-8")
data_raw = (out / "data.json").read_text(encoding="utf-8")
data = json.loads(data_raw)
results = {}

# A1 valid HTML: html5lib strict parse
try:
    doc = html5lib.HTMLParser(strict=True, namespaceHTMLElements=False).parse(html)
    ok = html.lstrip().lower().startswith("<!doctype html>")
    results["A1 fixture renders valid HTML"] = (ok, f"html5lib strict parse: no ParseError; doctype={ok}; bytes={len(html.encode())}")
except ParseError as e:
    results["A1 fixture renders valid HTML"] = (False, f"ParseError: {e}")
    doc = html5lib.parse(html, namespaceHTMLElements=False)

def text(el):
    return "".join(el.itertext())

# A2 bilingual crop: h1 holds Tamil (lang=ta) and English (lang=en) product names, Tamil first
h1 = doc.find(".//h1")
spans = [(s.get("lang"), text(s).strip()) for s in h1.iter("span")]
exp = [("ta", fx["product_name_local"]), ("en", fx["product_name_en"])]
ta_script = bool(re.search(r"[஀-௿]", fx["product_name_local"]))
title = text(doc.find(".//title")) if doc.find(".//title") is not None else ""
results["A2 bilingual crop name"] = (spans[:2] == exp and ta_script,
    f"h1 spans={spans}; expected={exp}; Tamil script in local name={ta_script}; <title>={title!r}")

# A3 units everywhere
def quantities(o, path="$"):
    if isinstance(o, dict):
        if set(o) >= {"value", "unit"} and not isinstance(o["value"], (dict, list)):
            yield path, o
        for k, v in o.items():
            yield from quantities(v, f"{path}.{k}")
    elif isinstance(o, list):
        for i, v in enumerate(o):
            yield from quantities(v, f"{path}[{i}]")
fq = list(quantities(fx)); dq = list(quantities(data))
shares = sum(1 for _ in re.finditer(r'"share_pct"', json.dumps(fx)))
dels = doc.findall(".//data")
bad_dom = [text(d) for d in dels if not re.fullmatch(r"[\d,.]+\s?\S+", text(d).strip()) or re.fullmatch(r"[\d,.]+", text(d).strip())]
no_unit_fx = [p for p, q in fq if not str(q["unit"]).strip()]
no_unit_dj = [p for p, q in dq if not str(q.get("unit") or "").strip()]
# Visible-text scan: numbers in dd/li/p that are bare (not inside <data>, <time>, ids, codes, counts)
expected_n = len(fq) + shares
results["A3 units on every quantity"] = (not bad_dom and not no_unit_fx and not no_unit_dj and len(dels) == expected_n,
    f"<data> elements={len(dels)}; fixture value+unit objects={len(fq)} + share_pct={shares} -> expected {expected_n}; "
    f"<data> without unit={bad_dom}; fixture qty without unit={no_unit_fx}; data.json qty without unit={no_unit_dj}; "
    f"sample={[text(d) for d in dels[:6]]}")

# A4 date order: every timeline <ol class=tl> has non-decreasing <time datetime>
lists = [ol for ol in doc.iter("ol") if "tl" in (ol.get("class") or "").split()]
viol, counts = [], []
for i, ol in enumerate(lists):
    ds = [t.get("datetime") for li in ol.findall("li") for t in li.findall(".//time")[:1]]
    counts.append(len(ds))
    if ds != sorted(ds):
        viol.append((i, ds))
# also data.json order
dj_viol = []
for l in data["lots"]:
    ds = [e["date"] for e in l["events"]]
    if ds != sorted(ds): dj_viol.append(l["lot_code"])
ds = [e["date"] for e in data.get("processing_events", [])]
if ds != sorted(ds): dj_viol.append("processing_events")
results["A4 date order"] = (bool(lists) and not viol and not dj_viol,
    f"timelines={len(lists)} events per timeline={counts}; HTML order violations={viol}; data.json violations={dj_viol}")

# A5 no temp_uploads / non-evidence paths
urls = set()
for el in doc.iter():
    for a in ("src", "href", "srcset", "data-src", "data-full", "poster"):
        v = el.get(a)
        if v:
            urls.update(u.split()[0] for u in v.split(","))
img_urls = sorted(u for u in urls if re.search(r"\.(jpe?g|webp|png|gif)$", u, re.I))
bad_img = [u for u in img_urls if "/evidence/" not in u]
json_urls = re.findall(r'https?://[^"\s]+', data_raw)
bad_json = [u for u in json_urls if re.search(r"\.(jpe?g|webp|png)$", u) and "/evidence/" not in u]
tmp_hits = [m for m in ("temp_uploads", "/tmp/", "tmp_uploads") for blob in (html, data_raw) if m in blob]
# boundary map is a static GI asset, not a photo: report separately
results["A5 no temp_uploads / non-evidence paths"] = (not bad_img or bad_img == [fx.get("boundary_map_url")]) and not tmp_hits and not [u for u in bad_json if u != fx.get("boundary_map_url")],
results["A5 no temp_uploads / non-evidence paths"] = (
    not tmp_hits and all(u == fx.get("boundary_map_url") for u in bad_img + bad_json),
    f"image URLs in HTML={len(img_urls)}; outside /evidence/ (excluding GI boundary_map_url static asset)="
    f"{[u for u in bad_img if u != fx.get('boundary_map_url')]}; data.json image URLs outside evidence="
    f"{[u for u in bad_json if u != fx.get('boundary_map_url')]}; temp markers found={tmp_hits}; "
    f"boundary_map_url (non-photo, https static)={fx.get('boundary_map_url')}")

fail = 0
for k, (ok, ev) in results.items():
    print(f"{'PASS' if ok else 'FAIL'} | {k} | {ev}")
    fail += not ok
print("exit", 1 if fail else 0)
sys.exit(1 if fail else 0)

"""List numbers in visible text not inside <data>, <time>, <a>, <style>, <script>."""
import re, sys, html5lib
doc = html5lib.parse(open(sys.argv[1], encoding="utf-8").read(), namespaceHTMLElements=False)
skip = {"data", "time", "style", "script", "title", "a"}
hits = []
def walk(el, inskip=False):
    s = inskip or el.tag in skip
    if not s and el.text: hits.extend(re.findall(r"[^\s<>]*\d[^\s<>]*", el.text))
    for c in el:
        walk(c, s)
        if not s and c.tail: hits.extend(re.findall(r"[^\s<>]*\d[^\s<>]*", c.tail))
walk(doc)
from collections import Counter
for k, v in sorted(Counter(hits).items()): print(v, k)
